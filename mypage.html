const container = document.getElementById('booking-list-container'); const completedBookings = JSON.parse(localStorage.getItem('completed_bookings') || '[]'); const lastBooking = JSON.parse(localStorage.getItem('last_booking_info')); if (completedBookings.length === 0 && !lastBooking) { container.innerHTML = `

예약 내역이 없습니다.

`; return; } container.innerHTML = ''; const all = [...completedBookings]; if (all.length === 0 && lastBooking) { renderBooking(container, { info: lastBooking, items: [] }); } else { all.reverse().forEach(b => renderBooking(container, b)); } }); function renderBooking(container, booking) { const info = booking.info; // items가 없는 경우 info.items를 사용 (최근 예약 대응) const items = booking.items && booking.items.length > 0 ? booking.items : (info.items || []); const date = booking.timestamp ? new Date(booking.timestamp).toLocaleDateString() : '최근 예약'; const card = document.createElement('div'); card.className = 'booking-card'; let itemsHtml = ''; items.forEach(item => { // 날짜 및 상세 정보 구성 let dateStr = item.date || '날짜 미지정'; if (item.name.includes('픽업샌딩')) { dateStr = `${item.pickupDate || '픽업일 미정'} ~ ${item.sendingDate || '샌딩일 미정'}`; } else if (item.name.includes('리조트 견적') && item.details) { dateStr = `${item.details.checkin} ~ ${item.details.checkout}`; } const timeStr = item.time ? ` | 시간: ${item.time}` : ''; const countStr = ` | 인원: ${item.count}명`; const detailStr = (item.details && typeof item.details === 'string') ? `
(${item.details})` : ''; // 통일된 금액 계산 로직: item.totalPrice 우선 const itemPrice = item.totalPrice || (item.price * item.count); itemsHtml += `
${item.name}
₩ ${itemPrice.toLocaleString()}
📅 ${dateStr}${timeStr}${countStr}${detailStr}
`; }); const requestsHtml = info.requests ? `
${info.requests}
` : ''; card.innerHTML = `

● 예약 내역 (${date})

${info.customerKorName}
${info.contact}
${info.reservationNumber || '-'}
${info.status || '확인중'}
${itemsHtml}
${requestsHtml}
총 결제 금액 ₩ ${(info.totalPrice || 0).toLocaleString()}
`; container.appendChild(card); }
